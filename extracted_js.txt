function updateRoomHeaderBadge(roomElement) {

let currentLatitude = null, currentLongitude = null, selectedPhotoName = null, selectedPhotoData = null;

function fileToCompressedDataURL(file, callback) {
const reader = new FileReader();
reader.onload = function(e) {
const img = new Image();
img.onload = function() {
const canvas = document.createElement('canvas');
const scale = Math.min(1, 600 / img.width);
canvas.width = Math.max(1, img.width * scale);
canvas.height = Math.max(1, img.height * scale);
canvas.getContext('2d').drawImage(img, 0, 0, canvas.width, canvas.height);
callback(canvas.toDataURL('image/jpeg', 0.7));
};
img.src = e.target.result;
};
reader.readAsDataURL(file);
}

const imageInput = document.getElementById('imageInput');
const photoPreviewContainer = document.getElementById('photoPreviewContainer');
const photoPreview = document.getElementById('photoPreview');

imageInput.addEventListener('change', function(e) {
const file = e.target.files[0];
if (file) {
selectedPhotoName = file.name;
fileToCompressedDataURL(file, data => { selectedPhotoData = data; });
const reader = new FileReader();
reader.onload = function(event) {
photoPreview.src = event.target.result;
photoPreviewContainer.style.display = "block";
};
reader.readAsDataURL(file);
}
});

function fetchLocation() {
const locationText = document.getElementById('locationText');
const locationStatus = document.getElementById('locationStatus');

if ("geolocation" in navigator) {
locationStatus.innerHTML = '<div class="location-dot"></div><span class="small fw-bold">GPS Status: Locating...</span>';
navigator.geolocation.getCurrentPosition((position) => {
currentLatitude = position.coords.latitude;
currentLongitude = position.coords.longitude;
locationText.innerHTML = `<strong>Lat:</strong> ${currentLatitude.toFixed(4)}, <strong>Long:</strong> ${currentLongitude.toFixed(4)} <br><a href="https://maps.google.com/?q=${currentLatitude},${currentLongitude}" target="_blank" class="fw-bold" style="color: #C9A961;"><i class="fa-solid fa-map-location-dot me-1"></i>View on Google Maps</a>`;
locationStatus.className = 'location-status active';
locationStatus.innerHTML = '<div class="location-dot"></div><span class="small fw-bold" style="color: #C9A961;">GPS Status: Active & Tracking</span>';
}, (error) => {
let errorMsg = "GPS error. ";
if (error.code === 1) errorMsg += "Permission denied. Please enable location access.";
else if (error.code === 2) errorMsg += "Position unavailable. Check your connection.";
else if (error.code === 3) errorMsg += "Timeout. Please try again.";
else errorMsg += "Unknown error occurred.";
locationText.innerText = errorMsg;
locationStatus.className = 'location-status error';
locationStatus.innerHTML = '<div class="location-dot" style="background: #c75050;"></div><span class="small fw-bold" style="color: #c75050;">GPS Status: Error</span>';
}, {
enableHighAccuracy: true,
timeout: 10000,
maximumAge: 0
});
} else {
locationText.innerText = "Geolocation not supported by your browser.";
locationStatus.className = 'location-status error';
locationStatus.innerHTML = '<div class="location-dot" style="background: #c75050;"></div><span class="small fw-bold" style="color: #c75050;">GPS Status: Not Supported</span>';
}
}

async function loadEntryLogs() {
try {
const response = await fetch('/api/entry-log');
const logs = await response.json();
const tbody = document.getElementById('gatepassLogs');
tbody.innerHTML = '';

logs.forEach((log, idx) => {
const time = new Date(log.created_at).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
const badgeClass = log.entry_type === 'IN' ? 'bg-success' : 'bg-warning text-dark';
const entryNo = log.entry_number || `ENT-${idx+1}`;
tbody.innerHTML += `<tr><td><small class="fw-bold" style="color: #C9A961;">${entryNo}</small></td><td>${log.student_name}</td><td>${log.room_no}</td><td><span class="badge ${badgeClass}">${log.entry_type}</span></td><td>${time}</td></tr>`;
});

if (logs.length >= 3) {
try {
const aiRes = await fetch('/api/ai-analyze-entries', {
method: 'POST', headers: { 'Content-Type': 'application/json' },
body: JSON.stringify({ logs: logs })
});
const aiData = await aiRes.json();
if (aiData.success) {
const box = document.getElementById('aiEntrySuggestion');
box.style.display = 'block';
document.getElementById('aiEntrySuggestionText').innerHTML = `<strong>Pattern:</strong> ${aiData.ai.pattern} | <strong>Peak:</strong> ${aiData.ai.peak_time}<br><strong>Insight:</strong> ${aiData.ai.insight}`;
}
} catch (err) { console.error('AI entry analysis failed', err); }
}
} catch (err) { console.error('Error loading entry logs:', err); }
}

async function loadContacts() {
try {
const response = await fetch('/api/emergency-contact');
const contacts = await response.json();
const contactsList = document.getElementById('contactsList');
const contactsList2 = document.getElementById('contactsList2');

contactsList.innerHTML = '<li class="list-group-item d-flex justify-content-between align-items-center px-0"><div><strong>Police Control Room:</strong> <span class="fw-bold ms-1"><i class="fa-solid fa-phone me-1" style="color: #C9A961;"></i>112 / 100</span></div><span class="badge" style="background: #C9A961; color: #0f0f0f;">Emergency</span></li>';

li.innerHTML = `<div><strong>${contact.contact_name}:</strong> <span class="fw-bold ms-1"><i class="fa-solid fa-phone me-1" style="color: #C9A961;"></i>${contact.phone_number}</span></div><button type="button" class="btn btn-sm btn-link p-0 text-decoration-none" style="color: #C9A961;" onclick="deleteContact('${contact._id}')">Remove</button>`;

});
} catch (err) { console.error('Error loading contacts:', err); }
}

function calculateSettlements() {
let debtors = [], creditors = [];

roommates.forEach(r => {
if (r.balance < -0.01) debtors.push({ name: r.name, phone: r.phone, amt: Math.abs(r.balance) });
else if (r.balance > 0.01) creditors.push({ name: r.name, phone: r.phone, amt: r.balance });
});

const container = document.getElementById('settlements-list'); container.innerHTML = "";

if (debtors.length === 0 && creditors.length === 0) {
container.innerHTML = `<small style="color:#C9A961; font-weight:bold;">All balances are cleared! 🎉</small>`; return;
}

let d = 0, c = 0;
while (d < debtors.length && c < creditors.length) {
let settleAmt = Math.min(debtors[d].amt, creditors[c].amt);
container.innerHTML += `<div class="settle-box"><b>${debtors[d].name}</b> ➔ <b>${creditors[c].name}</b> : <b>₹${settleAmt.toFixed(2)}</b></div>`;
debtors[d].amt -= settleAmt; creditors[c].amt -= settleAmt;
if (debtors[d].amt <= 0.01) d++; if (creditors[c].amt <= 0.01) c++;
}
}

// ==================== SHAKE & VOICE SOS ====================
let shakeDetectionActive = false;
let lastShakeTime = 0;
const SHAKE_THRESHOLD = 15;
const SHAKE_COOLDOWN = 5000;

function initShakeDetection() {
if (window.DeviceMotionEvent) {
window.addEventListener('devicemotion', handleShake, false);
shakeDetectionActive = true;
}
}

function handleShake(event) {
const acceleration = event.accelerationIncludingGravity;
const currentTime = new Date().getTime();

if ((currentTime - lastShakeTime) > SHAKE_COOLDOWN) {
const magnitude = Math.sqrt(acceleration.x * acceleration.x + acceleration.y * acceleration.y + acceleration.z * acceleration.z);
if (magnitude > SHAKE_THRESHOLD) {
lastShakeTime = currentTime;
triggerShakeSOS(magnitude);
}
}
}

async function startFaceRegistration() {

});

statusDiv.className = 'alert alert-success mt-2 small';
statusDiv.innerHTML = '<i class="fa-solid fa-check-circle me-2"></i><strong>Face registered successfully!</strong>';

if (registrationStream) registrationStream.getTracks().forEach(t => t.stop());
video.srcObject = null;
} else {
statusDiv.className = 'alert alert-warning mt-2 small';
statusDiv.innerHTML = 'No face detected. Center your face and ensure good lighting.';
}
} catch (err) {
statusDiv.className = 'alert alert-danger mt-2 small';
statusDiv.innerHTML = 'Camera permission required: ' + err.message;
}
}

async function startFaceEntry() {
await loadFaceModels();
const video = document.getElementById('entryVideo');
const statusDiv = document.getElementById('entryStatus');

try {
entryStream = await navigator.mediaDevices.getUserMedia({ video: { width: 640, height: 480 } });
video.srcObject = entryStream;

statusDiv.className = 'alert alert-info mt-2 small';
statusDiv.innerHTML = '<i class="fa-solid fa-spinner fa-spin me-2"></i>Scanning for face...';
statusDiv.classList.remove('d-none');

await new Promise(r => video.onloadedmetadata = () => r());

const detectorOptions = new faceapi.TinyFaceDetectorOptions({ inputSize: 320, scoreThreshold: 0.3 });
let isProcessing = false;

const detectFace = async () => {
if (!entryStream || !video.srcObject || isProcessing) return;

isProcessing = true;

try {
const detection = await faceapi.detectSingleFace(video, detectorOptions).withFaceLandmarks().withFaceDescriptor();

if (detection) {
const res = await fetch('/api/face-recognition', {
method: 'POST', headers: { 'Content-Type': 'application/json' },
body: JSON.stringify({ face_descriptor: Array.from(detection.descriptor) })
});
const result = await res.json();

if (result.matched) {
statusDiv.className = 'alert alert-success mt-2 small';
statusDiv.innerHTML = `<i class="fa-solid fa-check-circle me-2"></i><strong>Welcome, ${result.student.student_name}!</strong> Entry logged for Room ${result.student.room_no}`;
loadEntryLogs();
setTimeout(() => { if (entryStream) entryStream.getTracks().forEach(t => t.stop()); video.srcObject = null; }, 3000);
return;
}
}
} catch (err) {} finally { isProcessing = false; }

if (entryStream && video.srcObject) setTimeout(detectFace, 1500);
};

detectFace();
} catch (err) {
statusDiv.className = 'alert alert-danger mt-2 small';
statusDiv.innerHTML = 'Camera error: ' + err.message;
}
}
